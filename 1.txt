#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import rospy
import cv2
import numpy as np
import math

# 兼容性处理
np.float = np.float64  

import ros_numpy
from geometry_msgs.msg import Twist
from sensor_msgs.msg import Image, PointCloud2
from ultralytics import YOLO
from cv_bridge import CvBridge

class AutonomousFollowerSystem:
    def __init__(self):
        rospy.init_node("autonomous_fusion_controller")
        
        # 1. 动力学控制参数 (混合控制增益)
        self.target_dist = 1.5      # 目标保持距离 (m)
        self.linear_k = 0.4         # 纵向跟踪比例增益
        self.angular_p = 0.0025     # 目标横向跟踪比例增益
        self.angular_d = 0.008      # 目标横向跟踪微分增益
        self.lane_k = 0.0015        # 车道保持中心牵引力增益 (P控制)
        
        self.deadzone_x = 60        # 视觉控制死区 (px)
        self.img_center_x = 640     # 图像中心参考系
        self.last_error_x = 0.0     

        # 2. 人工势场法 (APF) 参数
        self.apf_k_att = 0.8        # 引力场增益
        self.apf_k_rep = 0.08       # 斥力场增益
        self.apf_rho_0 = 1.2        # 斥力场有效影响半径 (m)
        self.target_radius = 0.5    # 目标排斥过滤半径 (m)

        # 3. 状态机与数据关联参数
        self.is_target_locked = False
        self.locked_target_x = 0.0  
        self.locked_target_z = 0.0  
        self.tracking_threshold = 0.8 
        self.last_valid_cmd = Twist()
        self.last_seen_time = rospy.Time.now()
        self.retention_time = 0.5   

        # 4. 相机内参 (Extracted from Camera Info)
        self.fx = 609.15686
        self.fy = 609.10510
        self.cx = 641.17797
        self.cy = 364.35092

        # 5. 系统初始化
        rospy.loginfo("Initializing Multi-Objective Fusion Navigation System (YOLO + APF + Lane Keeping)...")
        self.model = YOLO("yolov8n.pt")
        self.bridge = CvBridge()
        self.latest_pc = None
        
        rospy.Subscriber("/ouster/points", PointCloud2, self.pc_callback, queue_size=1)
        rospy.Subscriber("/rgb/image_raw", Image, self.image_callback, queue_size=1, tcp_nodelay=True)
        
        self.cmd_pub = rospy.Publisher("/cmd_vel", Twist, queue_size=1) 
        self.debug_pub = rospy.Publisher("/follower/debug_image", Image, queue_size=1)
        
        rospy.loginfo("System initialized successfully. Monitoring active.")

    def pc_callback(self, msg):
        """处理 3D 激光雷达点云"""
        try:
            pc_data = ros_numpy.numpify(msg)
            mask = np.isfinite(pc_data['x']) & np.isfinite(pc_data['y']) & np.isfinite(pc_data['z'])
            x_lidar, y_lidar, z_lidar = pc_data['x'][mask], pc_data['y'][mask], pc_data['z'][mask]

            X_cam, Y_cam, Z_cam = -y_lidar, -z_lidar, x_lidar
            front_mask = Z_cam > 0.1
            self.latest_pc = np.vstack((X_cam[front_mask], Y_cam[front_mask], Z_cam[front_mask])).T
        except Exception as e:
            rospy.logerr(f"LiDAR data process exception: {e}")

    def extract_lane_center(self, frame):
        """
        基于形态学与像素分布的车道/可行驶区域检测
        提取图像底部感兴趣区域 (ROI) 的亮度质心，作为车道中心参考点。
        """
        height, width = frame.shape[:2]
        # 定义 ROI 为图像底部 1/3 区域 (去除非路面干扰)
        roi_top = int(height * 0.6)
        roi = frame[roi_top:height, :]
        
        # 转换为灰度图并进行自适应二值化 (提取高光车道线或明亮路面)
        gray = cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY)
        blur = cv2.GaussianBlur(gray, (5, 5), 0)
        _, thresh = cv2.threshold(blur, 160, 255, cv2.THRESH_BINARY)
        
        # 计算二值化图像的图像矩 (Image Moments) 以获取物理质心
        M = cv2.moments(thresh)
        if M["m00"] > 1000: # 确保提取到的有效路面像素数量充足
            lane_center_x = int(M["m10"] / M["m00"])
            lane_center_y = int(M["m01"] / M["m00"]) + roi_top
            return lane_center_x, lane_center_y, thresh
        else:
            return width // 2, height, thresh # 若无清晰标识，默认直行

    def image_callback(self, msg):
        """主控制节点：感知融合与多目标控制律解算"""
        try:
            frame = self.bridge.imgmsg_to_cv2(msg, "bgr8")
            current_time = rospy.Time.now()

            if self.latest_pc is None:
                rospy.logwarn_throttle(2, "Fault: LiDAR Point Cloud Unavailable.")
                cv2.putText(frame, "ERR: SENSOR SYNC FAULT", (30, 40), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)
                self.debug_pub.publish(self.bridge.cv2_to_imgmsg(frame, "bgr8"))
                return 

            points_3d = self.latest_pc.copy()
            u = (self.fx * points_3d[:, 0] / points_3d[:, 2]) + self.cx
            v = (self.fy * points_3d[:, 1] / points_3d[:, 2]) + self.cy

            # ==========================================
            # 模块 1：车道与可行驶区域检测
            # ==========================================
            lane_cx, lane_cy, lane_mask = self.extract_lane_center(frame)
            lane_error_x = self.img_center_x - lane_cx
            
            # 可视化车道引导点
            cv2.circle(frame, (lane_cx, lane_cy), 8, (255, 0, 255), -1)
            cv2.line(frame, (self.img_center_x, lane_cy), (lane_cx, lane_cy), (255, 0, 255), 2)

            # ==========================================
            # 模块 2：YOLO 目标推理与数据关联
            # ==========================================
            results = self.model(frame, imgsz=320, verbose=False)
            boxes = results[0].boxes
            
            twist = Twist() 
            target_detected = False
            candidates = []

            if boxes is not None and len(boxes) > 0:
                for box in boxes:
                    cls = int(box.cls[0].cpu().numpy())
                    if self.model.names[cls] == 'person':
                        conf = float(box.conf[0].cpu().numpy())
                        x1, y1, x2, y2 = map(int, box.xyxy[0].cpu().numpy())
                        center_x = (x1 + x2) / 2.0
                        
                        inside_box_mask = (u >= x1) & (u <= x2) & (v >= y1) & (v <= y2)
                        points_in_box = points_3d[inside_box_mask]

                        if len(points_in_box) > 10:
                            candidates.append({
                                'x': np.median(points_in_box[:, 0]),
                                'z': np.median(points_in_box[:, 2]),
                                'center_x_2d': center_x,
                                'bbox': (x1, y1, x2, y2),
                                'conf': conf
                            })

            best_candidate = None
            if len(candidates) > 0:
                if not self.is_target_locked:
                    candidates.sort(key=lambda c: c['z'])
                    best_candidate = candidates[0]
                    self.is_target_locked = True
                else:
                    min_distance = float('inf')
                    for cand in candidates:
                        euclidean_dist = math.hypot(cand['x'] - self.locked_target_x, cand['z'] - self.locked_target_z)
                        if euclidean_dist < min_distance and euclidean_dist < self.tracking_threshold:
                            min_distance = euclidean_dist
                            best_candidate = cand

            # ==========================================
            # 模块 3：环境 APF 计算 (斥力场)
            # ==========================================
            f_rep_x, f_rep_z = 0.0, 0.0
            target_x_ref = self.locked_target_x if best_candidate else 0.0
            target_z_ref = self.locked_target_z if best_candidate else 10.0
            
            dist_to_target_sq = (points_3d[:, 0] - target_x_ref)**2 + (points_3d[:, 2] - target_z_ref)**2
            env_mask = dist_to_target_sq > (self.target_radius**2)
            obs_points = points_3d[env_mask]
            
            if len(obs_points) > 0:
                distances = np.sqrt(obs_points[:, 0]**2 + obs_points[:, 2]**2)
                danger_mask = (distances < self.apf_rho_0) & (distances > 0.1)
                close_obs = obs_points[danger_mask]
                close_dist = distances[danger_mask]
                
                if len(close_obs) > 0:
                    magnitude = self.apf_k_rep * (1.0 / close_dist - 1.0 / self.apf_rho_0) * (1.0 / (close_dist**2))
                    f_rep_x = np.sum(magnitude * (-close_obs[:, 0] / close_dist))
                    f_rep_z = np.sum(magnitude * (-close_obs[:, 2] / close_dist))

            # ==========================================
            # 模块 4：多目标力场融合控制律 (Fusion Control)
            # ==========================================
            if best_candidate is not None:
                # 状态 A：锁定目标。融合目标引力、车道牵引力与障碍物斥力
                self.locked_target_x = best_candidate['x']
                self.locked_target_z = best_candidate['z']
                
                # 线速度：由目标引力主导，辅以斥力减速
                f_att_z = (self.locked_target_z - self.target_dist) * self.apf_k_att
                twist.linear.x = np.clip(f_att_z + f_rep_z, -0.3, 0.4)
                
                # 角速度：斥力 > 目标追踪误差 > 车道引导
                target_error_x = self.img_center_x - best_candidate['center_x_2d']
                
                if abs(f_rep_x) > 0.15:
                    twist.angular.z = np.clip(f_rep_x * 1.5, -0.6, 0.6)
                    cv2.putText(frame, "STATUS: APF EVASION", (30, 40), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 165, 255), 2)
                else:
                    target_pd = 0.0
                    if abs(target_error_x) > self.deadzone_x:
                        error_diff = target_error_x - self.last_error_x
                        target_pd = (target_error_x * self.angular_p) + (error_diff * self.angular_d)
                        self.last_error_x = target_error_x
                    else:
                        self.last_error_x = 0.0
                        
                    # 融合：70% 跟踪目标，30% 车道保持趋势
                    twist.angular.z = np.clip(target_pd * 0.7 + (lane_error_x * self.lane_k) * 0.3, -0.6, 0.6)
                    cv2.putText(frame, "STATUS: TRACKING & LANE ASSIST", (30, 40), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

                target_detected = True
                self.last_seen_time = current_time
                self.last_valid_cmd = twist
                
                bx1, by1, bx2, by2 = best_candidate['bbox']
                cv2.rectangle(frame, (bx1, by1), (bx2, by2), (0, 255, 0), 2)
                
            else:
                # 状态 B：目标丢失。进入车道保持巡航与搜索模式
                time_lost = (current_time - self.last_seen_time).to_sec()
                if time_lost < self.retention_time:
                    twist = self.last_valid_cmd 
                else:
                    self.is_target_locked = False
                    # 自主巡航控制律：仅受车道牵引力与环境斥力作用
                    cruise_speed = 0.2
                    twist.linear.x = np.clip(cruise_speed + f_rep_z, 0.0, 0.3)
                    
                    if abs(f_rep_x) > 0.15:
                        twist.angular.z = np.clip(f_rep_x * 1.5, -0.6, 0.6)
                    else:
                        twist.angular.z = np.clip(lane_error_x * self.lane_k, -0.4, 0.4)
                        
                    cv2.putText(frame, "STATUS: TARGET LOST - LANE CRUISING", (30, 40), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 0), 2)

            twist.angular.z = np.clip(twist.angular.z, -0.6, 0.6)
            self.cmd_pub.publish(twist)
            self.debug_pub.publish(self.bridge.cv2_to_imgmsg(frame, "bgr8"))
            
        except Exception as e:
            rospy.logerr(f"Algorithm node execution fault: {e}")

if __name__ == "__main__":
    try:
        AutonomousFollowerSystem()
        rospy.spin()
    except rospy.ROSInterruptException:
        pass
